int sum(int a, int b) {
    int result = 0;
    result = a + b;
    return result;
}

def maxFunc(int x, int y) {
    if (x > y) {
        return x;
    } else {
        return y;
    }
}

int factorial(int n) {
    int i = 1;
    int res = 1;
    while (i <= n) {
        res = res * i;
        i = i + 1;
    }
    return res;
}

def swap(int a, int b) {
    int temp = a;
    a = b;
    b = temp;
    return 0;
}

int main() {
    int x = 10;
    int y = 20;
    int z = sum(x, y);
    print(z);
    int m = maxFunc(x, y);
    print(m);
    int f = factorial(5);
    print(f);
    swap(x, y);
    if (x != y) {
        print(1);
    } else {
        print(0);
    }
    if (z >= 30) {
        print(100);
    }
    return 0;
}
