def sum(int a, int b) {
    int result; 
    result = a + b;
    return result;
}

def maxFunc(int x, int y) {
    if (x > y) {
        return x;
    } else {
        return y;
    }
}

def swap(int a, int b) {
    int temp;
    temp = a;
    a = b;
    b = temp;
    return;
}

def multiply(int x, int y) {
    int product;
    product = x * y;
    return product;
}
